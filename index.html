/** * ========================================================================= * MACRONEXUS STPM - GOOGLE APPS SCRIPT BACKEND * ========================================================================= */ /** * LANGKAH PENYELESAIAN RALAT "UrlFetchApp": * Sila pilih fungsi `setup_WAJIB_LARI_SEKALI` di menu atas (sebelah butang Debug) * dan tekan "Run". Luluskan kebenaran (Review Permissions -> Advanced -> Go to). */ function setup_WAJIB_LARI_SEKALI() { try { // Kod ini hanya bertindak memanggil API luar secara kosong untuk memaksa // Google mendaftarkan kebenaran (scope) 'script.external_request' kepada akaun anda. UrlFetchApp.fetch("https://www.google.com"); Logger.log("Kebenaran API luar telah berjaya disahkan!"); } catch (e) { Logger.log("Ralat Pengesahan: " + e.toString()); } } /** * Fungsi utama untuk menerima request POST dari web (Frontend) */ function doPost(e) { // Tetapan asal untuk sebarang ralat (fallback) let responseObj = { status: "error", message: "Tindakan tidak dikenali." }; try { // Parse data JSON yang dihantar oleh frontend const payload = JSON.parse(e.postData.contents); const action = payload.action; if (action === "fetch_all") { responseObj = handleFetchAll(); } else if (action === "generate_chapter") { responseObj = handleGenerateChapter(payload); } else if (action === "upload" || action === "saveQuestions") { responseObj = handleUpload(payload); } else if (action === "submit_score") { responseObj = handleSubmitScore(payload); } } catch (error) { responseObj = { status: "error", message: "Ralat Pelayan (GAS): " + error.toString() }; } // Pulangkan response dalam format JSON return ContentService.createTextOutput(JSON.stringify(responseObj)) .setMimeType(ContentService.MimeType.JSON); } /** * Mengendalikan permintaan GET (Berguna untuk semak sama ada link Web App hidup/aktif) */ function doGet(e) { return ContentService.createTextOutput("Backend MacroNexus STPM sedang beroperasi dengan lancar."); } /** * FUNGSI 1: Ambil semua soalan dari Google Sheet */ function handleFetchAll() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheets = ss.getSheets(); let allQuestions = []; sheets.forEach(sheet => { const sheetName = sheet.getName(); // Hanya baca tab yang mempunyai perkataan "Set" atau "Bank" if (sheetName.toLowerCase().includes("set") || sheetName.toLowerCase().includes("bank")) { const data = sheet.getDataRange().getValues(); if (data.length > 1) { // Ambil header baris pertama const headers = data[0].map(h => h.toString().toLowerCase().trim()); // Loop setiap baris soalan for (let i = 1; i < data.length; i++) { let row = data[i]; let qObj = { sheetName: sheetName }; headers.forEach((header, index) => { qObj[header] = row[index]; }); // Selaraskan format JSON untuk frontend let formattedQ = { sheetName: sheetName, chapter: qObj.chapter || qObj.bab || "1", q: qObj.question || qObj.soalan || "Soalan Kosong", opts: [ qObj.optiona || qObj.pilihan_a || qObj.option_a || "", qObj.optionb || qObj.pilihan_b || qObj.option_b || "", qObj.optionc || qObj.pilihan_c || qObj.option_c || "", qObj.optiond || qObj.pilihan_d || qObj.option_d || "" ], ans: qObj.correctanswer || qObj.jawapan || "A", explanation: qObj.explanation || qObj.penerangan || "Tiada penerangan." }; allQuestions.push(formattedQ); } } } }); return { status: "success", questions: allQuestions }; } /** * FUNGSI 2: Berkomunikasi dengan Gemini AI untuk Jana Soalan */ function handleGenerateChapter(payload) { const apiKey = payload.apiKey; const chapterId = payload.chapterId || "1"; const chapterName = payload.chapterName || ("Bab " + chapterId); // Model Flash adalah yang paling pantas dan stabil untuk teks berformat const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`; const systemPrompt = `Anda adalah pakar penilai ekonomi STPM. Bina 20 soalan aneka pilihan (KBAT) Makroekonomi STPM berdasarkan ${chapterName}. Kembalikan jawapan secara TEPAT dalam format JSON array: [{"Chapter": ${chapterId}, "Question": "...", "OptionA": "...", "OptionB": "...", "OptionC": "...", "OptionD": "...", "CorrectAnswer": "A", "Explanation": "..."}]. PASTIKAN RESPONSE ANDA ADALAH FORMAT JSON SAHAJA TANPA PENERANGAN TEKS ATAU TANDA BACKTICKS.`; const options = { method: "post", contentType: "application/json", payload: JSON.stringify({ contents: [{ parts: [{ text: `Bina 20 soalan (KBAT) Makroekonomi STPM untuk ${chapterName}.` }] }], systemInstruction: { parts: [{ text: systemPrompt }] }, generationConfig: { responseMimeType: "application/json" } }), muteHttpExceptions: true }; const res = UrlFetchApp.fetch(endpoint, options); if (res.getResponseCode() === 200) { const data = JSON.parse(res.getContentText()); let rawJson = data.candidates[0].content.parts[0].text; // Pembersihan JSON sekiranya AI terselit backticks rawJson = rawJson.replace(/```json/g, "").replace(/```/g, "").trim(); return { status: "success", data: JSON.parse(rawJson) }; } else { return { status: "error", message: "Gemini API Error: " + res.getContentText() }; } } /** * FUNGSI 3: Simpan / Upload soalan ke dalam Google Sheet (Tab Baharu) */ function handleUpload(payload) { let questions = payload.questionsData; // Fallback jika format dari frontend sedikit berbeza if (!questions && payload.questionsJson) { questions = JSON.parse(payload.questionsJson); } if (!questions || !questions.length) { return { status: "error", message: "Tiada data soalan diterima untuk disimpan." }; } const ss = SpreadsheetApp.getActiveSpreadsheet(); // Analisis nombor 'Set' tertinggi yang sedia ada let maxSet = 0; ss.getSheets().forEach(s => { let match = s.getName().match(/Set\s*(\d+)/i); if (match) maxSet = Math.max(maxSet, parseInt(match[1])); }); let nextSet = maxSet + 1; let newSheetName = "Set " + nextSet; let sheet = ss.getSheetByName(newSheetName); if (!sheet) { sheet = ss.insertSheet(newSheetName); // Masukkan Header Kolom standard sheet.appendRow(["QuestionID", "Chapter", "Topic", "Difficulty", "Question", "OptionA", "OptionB", "OptionC", "OptionD", "CorrectAnswer", "Explanation"]); sheet.getRange("A1:K1").setFontWeight("bold").setBackground("#f3f4f6"); // Serikan sikit warna } // Masukkan setiap soalan ke dalam baris Sheet questions.forEach(q => { sheet.appendRow([ q.QuestionID || `SET${nextSet}_B${q.Chapter||q.chapter||1}_${Math.floor(Math.random() * 9999)}`, q.Chapter || q.chapter || 1, q.Topic || "Janaan AI", q.Difficulty || "STPM", q.Question || q.question || "", q.OptionA || q.optionA || (q.opts ? q.opts[0] : ""), q.OptionB || q.optionB || (q.opts ? q.opts[1] : ""), q.OptionC || q.optionC || (q.opts ? q.opts[2] : ""), q.OptionD || q.optionD || (q.opts ? q.opts[3] : ""), q.CorrectAnswer || q.ans || "A", q.Explanation || q.explanation || "Tiada penerangan." ]); }); return { status: "success", message: `Berjaya menyimpan ${questions.length} soalan ke dalam ${newSheetName}` }; } /** * FUNGSI 4: (Opsyenal) Catat rekod markah pelajar di Google Sheet */ function handleSubmitScore(payload) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetName = "Rekod Skor"; let sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); sheet.appendRow(["Timestamp", "Nama Pelajar", "Kod Kelas", "Tahap (Level)", "Jumlah XP", "Soalan Dijawab", "Jawapan Tepat"]); sheet.getRange("A1:G1").setFontWeight("bold"); } sheet.appendRow([ new Date(), payload.name || "Pelajar Tidak Dikenali", payload.classCode || "Tiada", payload.level || 1, payload.xp || 0, payload.attempts || 0, payload.correct || 0 ]); return { status: "success", message: "Skor direkod." }; }
